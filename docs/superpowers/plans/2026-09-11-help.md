# Help Window Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `Clipjump.chm` and the scattered help entry points with one in-app WebView2 help window: task guides in front, a reference generated from the running app behind, and a filter across every topic.

**Architecture:** A new `src/ui/web/HelpWin.ahk` built on the existing `WebHost` class (the same host Clipboard History and the Channel Organizer use) shows `ui/help/index.html`. The page loads one HTML fragment per topic from `ui/help/topics/`, renders a topic list with a filter, and fills reference tables from a single `reference` message the host builds from live state (`Hud.HelpItems()`, `Formatting_Items("")`, `ActionRing_Items()`, `Rules.Actions`, `ClipType.Kinds`, the configured hotkey globals).

**Tech Stack:** AutoHotkey v2.0.27 (`runtime/AutoHotkey64.exe`), WebView2 through `lib/vendor/WebView2/WebView2.ahk` and `src/ui/web/WebHost.ahk`, vanilla JS/CSS with no build step (`ui/common/bridge.js`, `theme.css`, `util.js`, `mock-bridge.js`). Tests are AHK scripts under `tests/` run with `MSYS_NO_PATHCONV=1 ./runtime/AutoHotkey64.exe /ErrorStdOut tests/<name>.ahk` from Git Bash.

**Spec:** `docs/superpowers/specs/2026-09-11-help-design.md`

## Global Constraints

- AHK v2 only. A static field and a static method of one class must not share a name (case-insensitive). Functions that read globals declare them. `=` compares case-insensitively, `==` case-sensitively.
- Help prose is English only. New user-visible chrome strings go to `languages/english.txt` as `KEY = text`, read as `TXT.KEY`; the file is CRLF UTF-8 without a BOM, read under `FileEncoding "UTF-8"`.
- Web pages are vanilla JS and CSS with no build step and no external requests: everything loads from `https://app.clipjump/` (mapped to `ui/`). A page must work in Chrome against `ui/common/mock-bridge.js`, which is how layout work is done.
- Messages between page and host are `{cmd, data}` JSON both ways: page to host `bridge.send(cmd, data)`, host to page `host.Post(cmd, data)`. A page starts from `bridge.whenReady(fn)` and announces itself with `send("ready")`.
- File encodings and line endings stay as they are: `src/**/*.ahk`, `Clipjump.ahk` and `languages/english.txt` are CRLF; `ui/**` files are LF; `tests/*.ahk` are CRLF in the working tree. Verify with a byte check, not git-bash text tools.
- Validate (`MSYS_NO_PATHCONV=1 ./runtime/AutoHotkey64.exe /ErrorStdOut /validate Clipjump.ahk`) exits 0 with only the known noise: four WebView2 vendor warnings and one pre-existing SettingsWin warning.
- Interactive tests need an idle desktop and a running Clipjump restarted on the new code: `powershell -NoProfile -Command "Start-Process -FilePath '.\runtime\AutoHotkey64.exe' -ArgumentList '/ErrorStdOut','Clipjump.ahk'"`, wait 8 s. Wrap every AutoHotkey run in `timeout`. A headless run that hangs is usually an error or warning dialog: list processes with PowerShell `Get-CimInstance Win32_Process -Filter "Name='AutoHotkey64.exe'" | Select ProcessId, CommandLine` and kill only your own test process, never the one running `Clipjump.ahk`. Never drive Notepad. Never edit, stage or checkout `settings.ini`.
- WebView2 host-side handler errors are swallowed and logged: read `%TEMP%\Clipjump\webhost.log` when a message seems to do nothing. `WebHost.Log` truncates each line to 200 characters, so tests must not depend on late fields.
- Commit messages end with a blank line and `Co-Authored-By: Claude Opus 5 (1M context) <noreply@anthropic.com>`.

`src/ui/web/HelpWin.ahk` (CRLF, no BOM):

```ahk
; HelpWin.ahk - the Help window: a WebHost showing ui/help/index.html.
;
;   HelpWin.Open(topic)   HelpWin.Close()
;
; Page -> host: reference, open {url}, close
; Host -> page: i18n, reference {...}, opened {topic}

class HelpWin {
	static host := ""
	static section := "Help_window"
	static pending := ""

	static Instance() {
		global PROGNAME, TXT
		if !HelpWin.host {
			h := WebHost(PROGNAME " " TXT.HLP_title, "help/index.html", "Help")
			h.OnMessage(HelpWin_OnMessage)
			h.OnClose(HelpWin_SaveGeometry)
			HelpWin.host := h
		}
		return HelpWin.host
	}

	static Open(topic := "") {
		h := HelpWin.Instance()
		HelpWin.pending := topic
		if h.Visible {
			try WinActivate(h.g.Hwnd)
			if h.ready
				h.Post("opened", topic)
			return
		}
		if h.Show(HelpWin.Geometry()) {
			h.Focus()
			if h.ready
				h.Post("opened", topic)
		}
	}

	static Close() {
		if HelpWin.host && HelpWin.host.Visible {
			HelpWin_SaveGeometry()
			HelpWin.host.Hide()
		}
	}

	static Geometry() {
		global CONFIGURATION_FILE
		w := IniRead(CONFIGURATION_FILE, HelpWin.section, "w", 900), h := IniRead(CONFIGURATION_FILE, HelpWin.section, "h", 640)
		x := IniRead(CONFIGURATION_FILE, HelpWin.section, "x", ""), y := IniRead(CONFIGURATION_FILE, HelpWin.section, "y", "")
		w := (IsInteger(w) && w >= 520) ? w : 900, h := (IsInteger(h) && h >= 360) ? h : 640
		return ((IsInteger(x) && IsInteger(y)) ? "x" x " y" y " " : "") "w" w " h" h
	}

	static I18n() {
		global TXT
		keys := ["HLP_title", "HLP_search", "HLP_noMatch", "HLP_matches", "HLP_topics", "HLP_key", "HLP_action"
			, "HLP_format", "HLP_appliesTo", "HLP_shortcut", "HLP_description", "HLP_notAvailable"]
		m := Map()
		for k in keys
			m[k] := TXT.%k%
		HelpWin.host.Post("i18n", m)
	}
}

HelpWin_OnMessage(cmd, data, host) {
	switch cmd {
		case "ready":
			HelpWin.I18n()
			host.Post("reference", Help_Reference())
			host.Post("opened", HelpWin.pending)
		case "reference": host.Post("reference", Help_Reference())
		case "open": BrowserRun(data)
		case "close": HelpWin.Close()
	}
}

HelpWin_SaveGeometry() {
	h := HelpWin.host
	if !h || !h.Visible
		return
	try {
		WinGetPos(&x, &y, , , h.g.Hwnd)
		h.g.GetClientPos(, , &w, &hh)
		s := HelpWin.section
		Ini_write(s, "x", x, 0), Ini_write(s, "y", y, 0), Ini_write(s, "w", w, 0), Ini_write(s, "h", hh, 0)
	}
}

; the ACTIONS entry, so customizer scripts and ACTIONMODE can route to help by name
help(*) {
	HelpWin.Open()
}
```

`ui/help/app.js` (LF). This task renders the list and one topic; the filter arrives in Task 4 and the reference tables in Task 2.

```js
// app.js - the Help window page. Topics are HTML fragments under topics/; the reference tables inside them
// are filled from the host's `reference` message.
const TOPICS = [
  "start", "tasks", "reference-pastemode", "reference-actionmode", "reference-formats", "reference-rules",
  "reference-settings", "reference-plugins", "reference-shortcuts", "trouble", "about",
];
const state = { topics: [], current: "", i18n: {}, reference: null };

async function loadTopics() {
  const out = [];
  for (const id of TOPICS) {
    try {
      const res = await fetch("topics/" + id + ".html");
      if (!res.ok) continue;
      const el = document.createElement("div");
      el.innerHTML = await res.text();
      const root = el.firstElementChild;
      if (!root) continue;
      out.push({ id, title: root.dataset.title || id, keywords: root.dataset.keywords || "", node: root });
    } catch (e) { /* a missing fragment just drops out of the list */ }
  }
  return out;
}

function renderList() {
  const list = document.getElementById("list");
  list.textContent = "";
  for (const t of state.topics) {
    const b = document.createElement("button");
    b.textContent = t.title;
    b.dataset.id = t.id;
    if (t.id === state.current) b.className = "on";
    b.addEventListener("click", () => show(t.id));
    list.appendChild(b);
  }
}

function show(id) {
  const t = state.topics.find((x) => x.id === id) || state.topics[0];
  if (!t) return;
  state.current = t.id;
  const doc = document.getElementById("doc");
  doc.textContent = "";
  doc.appendChild(t.node);
  doc.scrollTop = 0;
  fillReference(doc);
  renderList();
}

function fillReference(root) { /* Task 2 */ }

bridge.on("i18n", (m) => {
  state.i18n = m || {};
  const q = document.getElementById("q");
  if (state.i18n.HLP_search) q.placeholder = state.i18n.HLP_search;
});
bridge.on("reference", (d) => { state.reference = d; fillReference(document.getElementById("doc")); });
bridge.on("opened", (topic) => { if (topic) show(topic); });

bridge.whenReady(async () => {
  state.topics = await loadTopics();
  show(state.topics.length ? state.topics[0].id : "");
  bridge.send("ready");
});
```

function fillReference(root) {
  if (!root || !state.reference) return;
  for (const el of root.querySelectorAll("[data-ref]")) {
    const rows = table(el.dataset.ref);
    el.textContent = "";
    if (!rows || !rows.length) {
      const p = document.createElement("p");
      p.className = "empty";
      p.textContent = state.i18n.HLP_notAvailable || "Not available in this build";
      el.appendChild(p);
      continue;
    }
    el.appendChild(buildTable(rows.head, rows.body));
  }
}

function table(name) {
  const r = state.reference || {};
  const t = state.i18n;
  const head = (a, b) => [t[a] || a, t[b] || b];
  switch (name) {
    case "pasteMode":
      return { head: head("HLP_key", "HLP_action"), body: (r.pasteMode || []).map((x) => [kbd(x.key), x.caption]) };
    case "actions":
      return { head: head("HLP_key", "HLP_action"), body: (r.actions || []).map((x) => [kbd(x.key), x.label + (x.desc ? " — " + x.desc : "")]) };
    case "formats":
      return {
        head: [t.HLP_key || "Key", t.HLP_format || "Format", t.HLP_appliesTo || "Applies to"],
        body: (r.formats || []).map((x) => [kbd(x.key), x.label + (x.desc ? " — " + x.desc : ""), x.kinds || "every clip"]),
      };
    case "hotkeys":
      return { head: head("HLP_shortcut", "HLP_description"), body: (r.hotkeys || []).map((x) => [kbd(x.keys), x.label]) };
    case "ruleActions":
      return { head: head("HLP_action", "HLP_description"), body: (r.rules && r.rules.actions || []).map((a) => [a, ""]) };
    case "kinds":
      return { head: head("HLP_key", "HLP_description"), body: (r.rules && r.rules.kinds || []).map((k) => [k.kind, k.label]) };
    default:
      return null;
  }
}

function kbd(s) { const k = document.createElement("kbd"); k.textContent = s; return k; }

function buildTable(head, body) {
  const tbl = document.createElement("table");
  const thead = document.createElement("thead");
  const tr = document.createElement("tr");
  for (const h of head) { const th = document.createElement("th"); th.textContent = h; tr.appendChild(th); }
  thead.appendChild(tr);
  tbl.appendChild(thead);
  const tb = document.createElement("tbody");
  for (const row of body) {
    const r = document.createElement("tr");
    for (const cell of row) {
      const td = document.createElement("td");
      if (cell instanceof Node) td.appendChild(cell); else td.textContent = cell;
      r.appendChild(td);
    }
    tb.appendChild(r);
  }
  tbl.appendChild(tb);
  return tbl;
}
